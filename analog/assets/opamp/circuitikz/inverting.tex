\documentclass[border=10pt]{standalone}
\usepackage[european,american voltages,american inductors,american currents]{circuitikz}
\ctikzset{resistors/scale=0.82,capacitors/scale=0.8,bipoles/length=1.2cm}
\begin{document}
\begin{circuitikz}[line width=0.8pt]
  \node[op amp] (op) at (4.5,0) {};
  \coordinate (inp) at ([xshift=-4cm]op.-);

  \draw (inp) node[left] {$v_i$}
    to[R=$R_1$] (op.-);

  \draw (op.+) -- (op.+ |- 0,-1.5) node[ground] {};

  \draw (op.-) -- ++(0,1.8) coordinate (fb-left)
    to[R=$R_f$] ++(3.2,0) coordinate (fb-right)
    -- (fb-right |- op.out)
    -- (op.out);

  \draw (op.out) -- ++(1.2,0) node[right] {$v_o$};

  \fill (op.-) circle (1.4pt);
  \fill (fb-right |- op.out) circle (1.4pt);
  \draw (op.-) node[above left=1pt] {$v_-$};
  \draw (op.+) node[below left=1pt] {$v_+$};
\end{circuitikz}
\end{document}
