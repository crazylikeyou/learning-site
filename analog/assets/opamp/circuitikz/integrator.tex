\documentclass[border=8pt]{standalone}
\usepackage{circuitikz}
\ctikzset{resistor=european,resistors/scale=0.8,capacitors/scale=0.8,bipoles/length=1.2cm}
\begin{document}
\begin{circuitikz}[line width=0.8pt]
  \node[op amp] (op) at (4,0) {};
  \coordinate (inv) at (op.-);

  \draw (0,0.45) node[left] {$v_i$} to[R=$R$] (inv);
  \draw (op.+) -- (op.+ |- 0,-1.5) node[ground] {};

  \draw (inv) -- ++(0,1.8) coordinate (fb-left)
    to[C=$C$] ++(3.2,0) coordinate (fb-right)
    -- (fb-right |- op.out)
    -- (op.out);

  \draw (op.out) -- ++(1.2,0) coordinate (out) node[right] {$v_o$};
  \fill (inv) circle (1.4pt);
  \fill (fb-right |- op.out) circle (1.4pt);
  \fill (out) circle (1.4pt);
  \draw (op.-) node[above left=1pt] {$v_-$};
  \draw (op.+) node[below left=1pt] {$v_+$};
\end{circuitikz}
\end{document}
