\documentclass[border=8pt]{standalone}
\usepackage{circuitikz}
\ctikzset{resistor=european,resistors/scale=0.8,capacitors/scale=0.8,bipoles/length=1.2cm}
\begin{document}
\begin{circuitikz}[line width=0.8pt]
  \node[op amp] (op) at (4,0) {};
  \coordinate (inp) at ([xshift=-3cm]op.+);
  \coordinate (invleft) at ([xshift=-1.0cm]op.-);
  \coordinate (fbbottom) at ([yshift=-1.7cm]op.-);
  \coordinate (corner) at (invleft |- fbbottom);

  \draw (inp) node[left] {$v_i$} -- (op.+);
  \draw (op.out) -- ++(1.3,0) coordinate (out) node[right] {$v_o$};
  \draw (out) -- (out |- fbbottom) -- (corner) -- (invleft) -- (op.-);
  \fill (op.-) circle (1.4pt);
  \fill (out) circle (1.4pt);
  \draw (op.-) node[above left=1pt] {$v_-$};
  \draw (op.+) node[below left=1pt] {$v_+$};
\end{circuitikz}
\end{document}
