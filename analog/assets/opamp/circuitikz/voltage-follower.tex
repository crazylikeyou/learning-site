\documentclass[border=10pt]{standalone}
\usepackage[european,american voltages,american inductors,american currents]{circuitikz}
\ctikzset{resistors/scale=0.82,capacitors/scale=0.8,bipoles/length=1.2cm}
\begin{document}
\begin{circuitikz}[line width=0.8pt]
  \node[op amp] (op) at (4.5,0) {};
  \coordinate (inp) at ([xshift=-3.8cm]op.+);
  \coordinate (fbleft) at ([yshift=1.6cm]op.-);
  \coordinate (fbright) at (op.out |- fbleft);
  \draw (inp) node[left] {$v_i$} -- (op.+);
  \draw (op.out) -- ++(1.2,0) coordinate (out) node[right] {$v_o$};

  \draw (op.out) -- (fbright) -- (fbleft) -- (op.-);

  \fill (op.-) circle (1.5pt);
  \fill (op.out) circle (1.5pt);
  \draw (op.-) node[below left=2pt] {$v_-$};
  \draw (op.+) node[below left=2pt] {$v_+$};
\end{circuitikz}
